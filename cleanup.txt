   215	        new_mems = []
   216	        combined_delta = torch.zeros_like(h_mamba)
   217	        
   218	        # Handle 2D vs 3D shapes for routing weights
   219	        # routing_weights shape can be [B, S, Ropes] or [B, Ropes] during inference/generation
   220	        rw = routing_weights if routing_weights.dim() == 3 else routing_weights.unsqueeze(1)
   221	        
   222	        for j in range(self.num_ropes):
   223	            rope_memory = memory[:, j]
   224	            specialist = self.specialist_grid[layer_idx][f"rope_{j}"]
   225	            sp_weight = rw[:, :, j].unsqueeze(-1)
   226	            
   227	            # Efficient skip for inference
   228	            if not self.training and sp_weight.max() < 1e-5:
   229	                new_mems.append(rope_memory)
   230	                continue
   231	                
   232	            h_out, m_out = specialist(h_mamba, rope_memory, weight=sp_weight)
   233	            combined_delta = combined_delta + sp_weight * (h_out - h_mamba)
   234	            new_mems.append(m_out)
   235	            
   236	        h_final = (h_mamba + combined_delta).to(dtype)
   237	        memory_final = torch.stack(new_mems, dim=1)
   238	        
   239	        # Stability losses
   240	        z_loss = torch.mean(torch.logsumexp(router_logits, dim=-1)**2)
   241	        freqs = routing_weights.mean(dim=(0, 1))
   242	        probs = torch.softmax(router_logits.mean(dim=(0, 1)), dim=-1)
   243	        aux_loss = self.num_ropes * torch.sum(freqs * probs)
   244	        total_loss = 0.01 * (z_loss * 0.1 + aux_loss)
   245	        
   246	        return h_final, memory_final, total_loss
